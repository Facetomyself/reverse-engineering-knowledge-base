```
def __feistel_L_R(self, L, R):
# 16轮feistel + 异或 + 交换for idx in range(16):
L ^= self.P[idx]
R ^= self.___feistel(L)
L, R = R, L
# 一轮异或 + 交换
L ^= self.P[16]
R ^= self.P[17]
L, R = R, L
return L, R
```

```
def ___feistel(self, x: int) -> int:
# 4个索引
a = (x >> 24) & 0xFF
b = (x >> 16) & 0xFF
c = (x >> 8) & 0xFF
d = x & 0xFF
y = (self.S[0][a] + self.S[1][b]) & 0xFFFFFFFFy ^= self.S[2][c]
```

# 二、OFB 模式

## 1. OFB 模式原理

OFB (Output Feedback) 将分组密码变成流密码:
●
加密解密操作相同
●
无需填充,密文长度 = 明文长度
●
IV 必须唯一,不可重复使用

## 2. OFB 加密流程

```
O[1] = Encrypt(O[0])  → C[0] = P[0] XOR O[1]
O[2] = Encrypt(O[1])  → C[1] = P[1] XOR O[2]
```

```
def encrypt_ofb(self, plaintext: bytes, key_bytes: bytes, iv: bytes) -> bytes:
"""Blowfish OFB 模式加密
Args:plaintext: 明文key_bytes: 密钥 (4-56 字节)iv: 初始化向量 (8 字节)
Returns:
密文 (不含 IV,长度 = 明文长度)
"""
self._key_expand(key_bytes)
ciphertext = b""
output_block = iv
for i in range(0, len(plaintext), 8):
# 加密输出块,生成密钥流
output_block = self._encrypt_block(output_block)
# 明文块与密钥流异或
plain_block = plaintext[i:i+8]
cipher_block = bytes(a ^ b for a, b in zip(plain_block, output_block))
ciphertext += cipher_block
return ciphertext
def decrypt_ofb(self, ciphertext: bytes, key_bytes: bytes, iv: bytes) -> bytes:
"""Blowfish OFB 模式解密 (与加密操作相同)"""
return self.encrypt_ofb(ciphertext, key_bytes, iv)
```
