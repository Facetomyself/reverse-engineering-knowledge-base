def extract_md5_blocks(log_text):
import re
# 正则解释:
# orr.*lsl #24 : 匹配你的搜索特征
# .*?\(w\)\s+w\d+=0x([0-9a-fA-F]+) : 忽略中间字符,捕获写入寄存器的十六进制值
pattern = re.compile(r"orr.*lsl #24.*?\(w\)\s+w\d+=0x([0-9a-fA-F]+)")
matches = pattern.findall(log_text)if not matches:print("未找到匹配的寄存器值")return
print(f"共找到 {len(matches)} 个寄存器写入值\n")
# 每 16 个 w 寄存器(4字节/个)构成一个 64 字节块
for i in range(0, len(matches), 16):
block_idx = i // 16 + 1
chunk_hex_list = matches[i:i+16]
if len(chunk_hex_list) < 16:
