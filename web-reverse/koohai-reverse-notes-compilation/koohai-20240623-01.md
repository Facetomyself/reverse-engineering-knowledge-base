          1. int __fastcall MD5Final(unsigned int *a1, unsigned __int8 *a2)

      2. {

      3.   unsigned int v3; // [sp+8h] [bp-28h]

      6.   7.   v4 = (*a1 >> 3) & 0x3F;

      8.   if ( v4 > 0x37 )

      9.     v3 = 120 - v4;

      10.   else

      11.     v3 = 56 - v4;

      12.   MD5Encode(v7, a1, 8u);

      14.   MD5Update(a1, v7, 8);
