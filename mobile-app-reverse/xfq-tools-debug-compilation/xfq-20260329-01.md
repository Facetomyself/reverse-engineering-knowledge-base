改系统语言设置的两种方案
1. adb shell 写配置 + reboot
adb shell "sx -c 'settings put system system_locales en-US'"
adb reboot
2. 直接frida注入system_server，用 frida -U -n system_server 然后输入下面的代码就行了
> 其实还有一种高级的玩法，是找到system提供的方法名号，然后service call，但是那个不稳定，主要是不兼容版本
Java.perform(function() {
    var Locale = Java.use("java.util.Locale");
    var targetLocale = Locale.$new("en", "US");
    // var targetLocale = Locale.$new("zh", "CN");

    var ActivityManagerNative = Java.use("android.app.ActivityManagerNative");
    var ams = ActivityManagerNative.getDefault();

    var config = ams.getConfiguration();

    config.locale.value = targetLocale;
    config.userSetLocale.value = true; 

    ams.updateConfiguration(config);
});
