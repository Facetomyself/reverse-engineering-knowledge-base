## 正文

没啥用的小知识，lsp拦截accounts&passowrds下的三方app的登录信息记录
Class AccountManagerClass = YQHelpers.findClass("android.accounts.AccountManager", loadPackageParam.classLoader);
YQHelpers.findAndHookMethod(AccountManagerClass, "addAccountExplicitly",
        Account.class, String.class, Bundle.class,
        new YQMethodHook() {
            @Override
            protected void beforeHookedMethod(YQMethodParam param) throws Throwable {
                Account account = (Account) param.args[0];
                String password = (String) param.args[1];
                Bundle userdata = (Bundle) param.args[2];
                // 阻止账户写入：设置方法返回结果为false
                param.setResult(false);
                YQHelpers.log("AccountServiceHook: 拦截账户添加: " + account.name + " " + account.type + " " + password + " " + userdata);
            }
        }
);
