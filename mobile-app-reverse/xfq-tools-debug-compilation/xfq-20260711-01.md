scrcpy常用命令
更全面的建议看官方文档，下面我只列一些常见的：[scrcpy/doc at master · Genymobile/scrcpy · GitHub](https://github.com/Genymobile/scrcpy/tree/master/doc)

# 普通投屏
 scrcpy -s 35051FDH2000NQ

  # 只录制，不弹窗口
  scrcpy -s 35051FDH2000NQ --no-window --record out.mp4

  # 录制画面 + 系统播放声音，手机也继续出声
  scrcpy -s 35051FDH2000NQ \
    --no-window \
    --record out.mp4 \
    --audio-source=playback \
    --audio-dup \
    --audio-codec=aac

  # 强制要求音频可用；没音频就失败
  scrcpy -s 35051FDH2000NQ \
    --no-window \
    --record out.mp4 \
    --audio-source=playback \
    --audio-dup \
    --audio-codec=aac \
    --require-audio

  # 投屏但不录制
  scrcpy -s 35051FDH2000NQ --audio-source=playback --audio-dup

  # 只录音频，不录画面
  scrcpy -s 35051FDH2000NQ \
    --no-video \
    --record audio.m4a \
    --record-format=m4a \
    --audio-source=playback \
    --audio-codec=aac

  # 降低分辨率，减少卡顿
  scrcpy -s 35051FDH2000NQ --max-size=1280

  # 设置视频码率
  scrcpy -s 35051FDH2000NQ --video-bit-rate=8M

  # 设置帧率
  scrcpy -s 35051FDH2000NQ --max-fps=30

  # 关闭电脑端播放，只保留录制
  scrcpy -s 35051FDH2000NQ --no-playback --record out.mp4

  # 查看设备 displays
  scrcpy -s 35051FDH2000NQ --list-displays

  # 指定 display
  scrcpy -s 35051FDH2000NQ --display-id=0

  # 窗口置顶
  scrcpy -s 35051FDH2000NQ --always-on-top

  # 录制 mkv，兼容性有时比 mp4 更稳
  scrcpy -s 35051FDH2000NQ \
    --no-window \
    --record out.mkv \
    --record-format=mkv \
    --audio-source=playback \
    --audio-dup

  我建议你最常用这两个：

  # 看手机
  scrcpy -s 35051FDH2000NQ

  # 自动化录证据，画面+声音
  scrcpy -s 35051FDH2000NQ --no-window --record out.mp4 \
    --audio-source=playback --audio-dup --audio-codec=aac --require-audio
